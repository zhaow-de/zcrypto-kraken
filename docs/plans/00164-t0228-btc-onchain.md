from pathlib import Path

def pooled(name, series):
    """A cell's pooled row of one family, `None` where the cell keeps none."""
    return m["cells"].get(name, {}).get(series, {}).get("pooled")


def years(name):
    return sorted(y for y, n in m["event_years"].get(name, {}).items() if n > 0)


checks = {
    "labels": (r["reading_id"], r["iteration"], r["spec_hash"], r["attempt"]) == (int(sys.argv[2]), sys.argv[3], sys.argv[1], None),
    "keys": set(m) == SUBJECTS["c3-onchain-reading"].metric_keys,

          far_high, "| action", action, "| summed mean after the events", after, "| legs 1, 2, 3, 4", legs, "| potential", all(legs))
