"""The C3 on-chain reading off the ledger's last line: its checks against R3's basis, then, every check holding, the table
spec 00164 D12's criterion is read off — one line per cell the floor admits, its four legs' values and whether each
holds, then the counts; exit 0 only when every check holds. Arguments: the spec's sha256, the reading's id, its
iteration."""

import json
import sys
from pathlib import Path

from cli.alpha.c3 import FEATURES, HORIZONS, STATES, TAILS, cell_name
from cli.research.subjects import SUBJECTS
from tests.test_research_c3_onchain_reading_basis import ARM_BASIS, COUNT_BASIS, FEATURE_BASIS

CONSTANTS = {
    "low": 0.10,
    "high": 0.90,
    "min_history": 365,
    "lag_days": 1,
    "horizons": [7, 30],
    "mean_block": 102,
    "n_resamples": 10000,
    "seed": 42,
    "far": [0.078125, 99.921875],
    "cost_round_trip": 0.012,
    "min_event_years": 4,
}
r = json.loads(Path("docs/reference/reading-ledger.jsonl").read_text().splitlines()[-1])
m = r["metrics"]
names = {cell_name(f, t, s, h) for f in FEATURES for t in TAILS for s in STATES for h in HORIZONS}


def pooled(name, series):
    """A cell's pooled row of one family, `None` where the cell keeps none."""
    return m["cells"].get(name, {}).get(series, {}).get("pooled")


def years(name):
    return sorted(y for y, n in m["event_years"].get(name, {}).items() if n > 0)


checks = {
    "labels": (r["reading_id"], r["iteration"], r["spec_hash"], r["attempt"]) == (int(sys.argv[2]), sys.argv[3], sys.argv[1], None),
    "keys": set(m) == SUBJECTS["c3-onchain-reading"].metric_keys,
    "constants": m["constants"] == CONSTANTS,
    "features": m["features"] == FEATURE_BASIS,
    "arm": m["arm"] == ARM_BASIS,
    "counts": m["counts"] == COUNT_BASIS,
    "cells": set(m["cells"]) == names and set(m["event_years"]) <= names,
    "event_years": all(
        sum(m["event_years"].get(n, {}).values()) == pooled(n, "log")[0] for n in names if pooled(n, "log") is not None
    ),
}
print(r["reading_id"], checks)
if not all(checks.values()):
    sys.exit(1)
cost, floor = m["constants"]["cost_round_trip"], m["constants"]["min_event_years"]
admitted, held = [], [0, 0, 0, 0]
for name in sorted(names):
    log, summed = pooled(name, "log"), pooled(name, "summed")
    if log is None or len(years(name)) < floor:
        continue
    state, diff, far_low, far_high = name.split("|")[2], log[4], log[7], log[8]
    action = "de-risk" if state == "long" and diff < 0 else "re-entry" if state == "flat" and diff > 0 else None
    after = None if summed is None else summed[2]
    leg4 = after is not None and (after <= -cost if action == "de-risk" else action == "re-entry" and after >= cost)
    legs = (far_low > 0 or far_high < 0, abs(diff) >= cost, action is not None, leg4)
    admitted.append((name, all(legs)))
    held = [h + leg for h, leg in zip(held, legs, strict=True)]
    print(name, "| event years", years(name), "| events, others", log[0], log[1], "| difference", diff, "| far bounds", far_low,
          far_high, "| action", action, "| summed mean after the events", after, "| legs 1, 2, 3, 4", legs, "| potential", all(legs))
print("cells", len(names), "| the floor admits", len(admitted), "| hold legs 1, 2, 3, 4", held, "| potential",
      [n for n, p in admitted if p] or "none")
